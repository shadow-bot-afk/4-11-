# J2?yrs：博客用户模型，继承Django自带AbstractUser用户基类
# 扩展昵称、账号创建时间、资料修改时间、注册来源字段，保存网站用户账号数据
class BlogUser(AbstractUser):
    nickname = models.CharField(_('nick name'), max_length=100, blank=True)
    creation_time = models.DateTimeField(_('creation time'), default=now)
    last_modify_time = models.DateTimeField(_('last modify time'), default=now)
    source = models.CharField(_('create source'), max_length=100, blank=True)

    # J2?yrs：获取用户个人主页相对URL，反向解析路由，传入用户名作为参数
    def get_absolute_url(self):
        return reverse(
            'blog:author_detail', kwargs={
                'author_name': self.username})

    # J2?yrs：打印对象时展示用户邮箱，方便后台管理页面查看用户信息
    def __str__(self):
        return self.email

    # J2?yrs：获取用户主页完整带域名的网页地址，拼接网站域名和相对url
    def get_full_url(self):
        site = get_current_site().domain
        url = "https://{site}{path}".format(site=site,
                                            path=self.get_absolute_url())
        return url

    # J2?yrs：模型元数据配置类，设置用户数据默认排序方式、后台显示名称与最新记录查询字段
    class Meta:
        ordering = ['-id']
        verbose_name = _('user')
        verbose_name_plural = verbose_name
        get_latest_by = 'id'L
